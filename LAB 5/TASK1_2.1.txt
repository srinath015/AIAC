import getpass

def collect_user_data():
    """
    Collects user data (name, age, email) and protects access with a password.
    Displays the collected data with the email partially hidden.
    """
    # Prompt for password (input is hidden)
    password = getpass.getpass("Set a password to protect your data: ")

    # Collect user details
    name = input("Enter your name: ")
    age = input("Enter your age: ")
    email = input("Enter your email: ")

    # Ask for password to show data
    entered_password = getpass.getpass("Enter your password to view your data: ")
    if entered_password != password:
        print("Incorrect password. Access denied.")
        return

    # Hide the email username part
    if "@" in email:
        hidden_email = "*****" + email[email.find("@"):]
    else:
        hidden_email = "*****"

    # Show collected data with comments
    print("\nCollected User Data:")
    print(f"# Name of the user")
    print(f"Name: {name}")
    print(f"# Age of the user")
    print(f"Age: {age}")
    print(f"# Email of the user (username hidden for privacy)")
    print(f"Email: {hidden_email}")

if __name__ == "__main__":
    collect_user_data()
# This code collects user data and protects access with a password.
